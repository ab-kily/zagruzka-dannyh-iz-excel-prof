&НаКлиенте
Процедура ПротоколЗагрузкиВыбор(Элемент, ВыбраннаяСтрока, Поле, СтандартнаяОбработка)

		Возврат;
	КонецЕсли;
	
	СтандартнаяОбработка = Ложь;
	ПерейтиПоНавигационнойСсылке(ТекущиеДанные.НавигационнаяСсылка);

#Область ЗагрузкаПорциями

&НаКлиенте
Процедура НачатьЗагрузку()
	
	Если ПустаяСтрока(АдресДанныхФайла) Тогда
		ПоказатьПредупреждение(, "Сначала прочитайте файл.");
		Возврат;
	КонецЕсли;
	Если ПустаяСтрока(ИмяОбъектаМетаданных) Тогда
		ПоказатьПредупреждение(, "Выберите объект загрузки.");
		Возврат;
	КонецЕсли;
	
	ПротоколЗагрузки.Очистить();
	СозданоОбъектов = 0;
	ОбновленоОбъектов = 0;
	СтрокСОшибками = 0;
	ПропущеноСтрок = 0;
	ИндексСледующейСтроки = 0;
	ИндикаторПрогресса = 0;
	ЗагрузкаИдет = Истина;
	Элементы.ИндикаторПрогресса.МаксимальноеЗначение = Макс(ВсегоСтрокФайла, 1);
	
	ИнициализироватьКэшСсылокНаСервере();
	ПодключитьОбработчикОжидания("ЗагрузитьСледующуюПорцию", 0.1, Истина);
	
КонецПроцедуры

&НаКлиенте
Процедура ЗагрузитьСледующуюПорцию()
	
	Если Не ЗагрузкаИдет Тогда
		Возврат;
	КонецЕсли;
	
	Результат = ЗагрузитьПорциюНаСервере();
	
	СозданоОбъектов = СозданоОбъектов + Результат.Создано;
	ОбновленоОбъектов = ОбновленоОбъектов + Результат.Обновлено;
	СтрокСОшибками = СтрокСОшибками + Результат.Ошибок;
	ПропущеноСтрок = ПропущеноСтрок + Результат.Пропущено;
	ИндикаторПрогресса = Мин(ВсегоСтрокФайла, ИндексСледующейСтроки);
	
	Если Результат.Прервано Или ИндексСледующейСтроки >= ВсегоСтрокФайла Тогда
		ЗагрузкаИдет = Ложь;
		ТекстИтога = СтрШаблон(
			"Готово. Создано: %1, обновлено: %2, ошибок: %3, пропущено: %4",
			СозданоОбъектов,
			ОбновленоОбъектов,
			СтрокСОшибками,
			ПропущеноСтрок);
		Если Результат.Прервано Тогда
			ПоказатьПредупреждение(, Результат.СообщениеПрерывания + Символы.ПС + ТекстИтога);
		Иначе
			ПоказатьОповещениеПользователя("Загрузка завершена", , ТекстИтога);
		КонецЕсли;
		Возврат;
	КонецЕсли;
	
	ПодключитьОбработчикОжидания("ЗагрузитьСледующуюПорцию", 0.1, Истина);
	
КонецПроцедуры

&НаСервере
Функция ЗагрузитьПорциюНаСервере()
	
	ОбработкаОбъект = РеквизитФормыВЗначение("Объект");
	ПараметрыЗагрузки = ПараметрыЗагрузкиДляОбъекта();
	ПараметрыЗагрузки.Вставить("НачальнаяСтрока", ИндексСледующейСтроки);
	ПараметрыЗагрузки.Вставить("КоличествоСтрок", РазмерПорции);
	
	Результат = ОбработкаОбъект.ВыполнитьЗагрузкуПорции(АдресДанныхФайла, АдресКэшаСсылок, ПараметрыЗагрузки);
	
	Для Каждого СтрокаПротокола Из Результат.Протокол Цикл
		НоваяСтрока = ПротоколЗагрузки.Добавить();
		ЗаполнитьЗначенияСвойств(НоваяСтрока, СтрокаПротокола);
	КонецЦикла;
	
	ИндексСледующейСтроки = ИндексСледующейСтроки + ПараметрыЗагрузки.КоличествоСтрок;
	Если ИндексСледующейСтроки > ВсегоСтрокФайла Тогда
		ИндексСледующейСтроки = ВсегоСтрокФайла;
	КонецЕсли;
	
	Возврат Результат;
	
КонецФункции

&НаСервере
Процедура ИнициализироватьКэшСсылокНаСервере()
	
	АдресКэшаСсылок = ПоместитьВоВременноеХранилище(Новый Соответствие, УникальныйИдентификатор);
	
КонецПроцедуры

&НаСервере
Функция ПараметрыЗагрузкиДляОбъекта()
	
	Маппинг = Новый Массив;
	Для Каждого СтрокаТаблицы Из МаппингКолонок Цикл
		ИмяРеквизита = СтрокаТаблицы.ИмяРеквизита;
		ИмяТЧ = СтрокаТаблицы.ИмяТабличнойЧасти;
		Если СтрНайти(ИмяРеквизита, ".") > 0 Тогда
			ЧастиИмени = СтрРазделить(ИмяРеквизита, ".");
			ИмяТЧ = ЧастиИмени[0];
			ИмяРеквизита = ЧастиИмени[1];
		КонецЕсли;
		Элемент = Новый Структура;
		Элемент.Вставить("ИмяКолонкиФайла", СтрокаТаблицы.ИмяКолонкиФайла);
		Элемент.Вставить("ЗаголовокКолонкиФайла", СтрокаТаблицы.ЗаголовокКолонкиФайла);
		Элемент.Вставить("ИмяРеквизита", ИмяРеквизита);
		Элемент.Вставить("ЭтоСсылка", СтрокаТаблицы.ЭтоСсылка);
		Элемент.Вставить("ПолноеИмяСсылочногоТипа", СтрокаТаблицы.ПолноеИмяСсылочногоТипа);
		Элемент.Вставить("ИскатьСуществующий", СтрокаТаблицы.ИскатьСуществующий);
		Элемент.Вставить("ВсегдаСоздаватьНовый", СтрокаТаблицы.ВсегдаСоздаватьНовый);
		Элемент.Вставить("ИмяРеквизитаПоискаСсылки", СтрокаТаблицы.ИмяРеквизитаПоискаСсылки);
		Элемент.Вставить("ДействиеЕслиНеНайден", СтрокаТаблицы.ДействиеЕслиНеНайден);
		Элемент.Вставить("РазбиратьСоставнойКлюч", СтрокаТаблицы.РазбиратьСоставнойКлюч);
		Элемент.Вставить("РазделительКлюча", СтрокаТаблицы.РазделительКлюча);
		Элемент.Вставить("ИмяТабличнойЧасти", ИмяТЧ);
		Маппинг.Добавить(Элемент);
	КонецЦикла;
	
	ПараметрыЗагрузки = Новый Структура;
	ПараметрыЗагрузки.Вставить("ТипДанных", ТипДанных);
	ПараметрыЗагрузки.Вставить("ИмяОбъектаМетаданных", ИмяОбъектаМетаданных);
	ПараметрыЗагрузки.Вставить("Маппинг", Маппинг);
	ПараметрыЗагрузки.Вставить("ПрерыватьПриОшибке", ПрерыватьПриОшибке);
	ПараметрыЗагрузки.Вставить("РазделительСоставногоКлюча", РазделительСоставногоКлюча);
	Возврат ПараметрыЗагрузки;
	
КонецФункции

&НаСервере
Функция ПроверитьГотовностьЗагрузкиНаСервере()
	
	Если ПустаяСтрока(АдресДанныхФайла) Тогда
		Возврат "Файл не прочитан.";
	КонецЕсли;
	Если ПустаяСтрока(ТипДанных) Или ПустаяСтрока(ИмяОбъектаМетаданных) Тогда
		Возврат "Не выбран объект загрузки.";
	КонецЕсли;
	
	ЕстьХотяБыОдноСоответствие = Ложь;
	Для Каждого СтрокаМаппинга Из МаппингКолонок Цикл
		Если Не ПустаяСтрока(СтрокаМаппинга.ИмяРеквизита) Тогда
			ЕстьХотяБыОдноСоответствие = Истина;
			Прервать;
		КонецЕсли;
	КонецЦикла;
	Если Не ЕстьХотяБыОдноСоответствие Тогда
		Возврат "Не задано ни одного соответствия колонок.";
	КонецЕсли;
	
	Незамапленные = Новый Массив;
	Для Каждого СтрокаРеквизита Из РеквизитыОбъекта Цикл
		Если Не СтрокаРеквизита.ПроверкаЗаполнения Или Не ПустаяСтрока(СтрокаРеквизита.ИмяТабличнойЧасти) Тогда
			Продолжить;
		КонецЕсли;
		КлючРеквизита = СтрокаРеквизита.Имя;
		Если Не ПустаяСтрока(СтрокаРеквизита.ИмяТабличнойЧасти) Тогда
			КлючРеквизита = СтрокаРеквизита.ИмяТабличнойЧасти + "." + СтрокаРеквизита.Имя;
		КонецЕсли;
		Найден = Ложь;
		Для Каждого СтрокаМаппинга Из МаппингКолонок Цикл
			Если СтрокаМаппинга.ИмяРеквизита = СтрокаРеквизита.Имя
				Или СтрокаМаппинга.ИмяРеквизита = КлючРеквизита Тогда
				Найден = Истина;
				Прервать;
			КонецЕсли;
		КонецЦикла;
		Если Не Найден Тогда
			Незамапленные.Добавить(СтрокаРеквизита.Синоним);
		КонецЕсли;
	КонецЦикла;
	
	Если Незамапленные.Количество() = 0 Тогда
		Возврат "";
	КонецЕсли;
	
	Возврат "Не сопоставлены обязательные реквизиты: " + СтрСоединить(Незамапленные, ", ") + ".";
	
КонецФункции

&НаКлиенте
Процедура ПрочитатьВыбранныйФайл(ПоказыватьПредупреждение)
	
	Если ПустаяСтрока(АдресФайла) Тогда
		Если ПоказыватьПредупреждение Тогда
			ПоказатьПредупреждение(, "Сначала выберите файл.");
		КонецЕсли;
		Возврат;
	КонецЕсли;
	
	ПрочитатьФайлНаСервере();
	
	Если ЗначениеЗаполнено(ИмяОбъектаМетаданных) Тогда
		ЗаполнитьМаппингПоОбъектуНаСервере();
	КонецЕсли;
	
КонецПроцедуры

&НаСервере
Процедура ПрочитатьФайлНаСервере()
	
	ОбработкаОбъект = РеквизитФормыВЗначение("Объект");
	Результат = ОбработкаОбъект.ПрочитатьФайлЗагрузки(
		АдресФайла,
		РасширениеФайла,
		ИмяЛиста,
		ПерваяСтрокаЗаголовки,
		РазделительCSV);
	
	АдресДанныхФайла = ПоместитьВоВременноеХранилище(Результат, УникальныйИдентификатор);
	ВсегоСтрокФайла = Результат.ВсегоСтрок;
	
	Если Результат.ИменаЛистов.Количество() > 0 Тогда
		Элементы.ИмяЛиста.СписокВыбора.Очистить();
		Для Каждого Имя Из Результат.ИменаЛистов Цикл
			Элементы.ИмяЛиста.СписокВыбора.Добавить(Имя);

&НаСервере
Процедура ЗаполнитьСтрокуМаппингаПоРеквизитуНаСервере(ИдентификаторСтроки)
	
	СтрокаМаппинга = МаппингКолонок.НайтиПоИдентификатору(ИдентификаторСтроки);
	Если СтрокаМаппинга = Неопределено Тогда
		Возврат;
	КонецЕсли;
	
	ИмяРеквизита = СтрокаМаппинга.ИмяРеквизита;
	ИмяТЧ = "";
	Если СтрНайти(ИмяРеквизита, ".") > 0 Тогда
		Части = СтрРазделить(ИмяРеквизита, ".");
		ИмяТЧ = Части[0];
		ИмяРеквизита = Части[1];
	КонецЕсли;
	СтрокаМаппинга.ИмяТабличнойЧасти = ИмяТЧ;
	
	Для Каждого СтрокаРеквизита Из РеквизитыОбъекта Цикл
		Если СтрокаРеквизита.Имя = ИмяРеквизита И СтрокаРеквизита.ИмяТабличнойЧасти = ИмяТЧ Тогда
			СтрокаМаппинга.ПредставлениеРеквизита = СтрокаРеквизита.Синоним;
			СтрокаМаппинга.ТипПредставление = СтрокаРеквизита.ТипПредставление;
			СтрокаМаппинга.ЭтоСсылка = СтрокаРеквизита.ЭтоСсылка;
			СтрокаМаппинга.ПолноеИмяСсылочногоТипа = СтрокаРеквизита.ПолноеИмяСсылочногоТипа;
			СтрокаМаппинга.Обязательный = СтрокаРеквизита.ПроверкаЗаполнения;
			СтрокаМаппинга.ИменаРеквизитовПоиска = СтрокаРеквизита.ИменаРеквизитовПоиска;
			Если СтрокаРеквизита.ЭтоСсылка И ПустаяСтрока(СтрокаМаппинга.ИмяРеквизитаПоискаСсылки) Тогда
				ИменаПоиска = СтрРазделить(СтрокаРеквизита.ИменаРеквизитовПоиска, ",", Ложь);
				Если ИменаПоиска.Количество() > 0 Тогда
					СтрокаМаппинга.ИмяРеквизитаПоискаСсылки = СокрЛП(ИменаПоиска[0]);
				КонецЕсли;
				Если ПустаяСтрока(СтрокаМаппинга.ДействиеЕслиНеНайден) Тогда
					СтрокаМаппинга.ДействиеЕслиНеНайден = "ОставитьПустым";
				КонецЕсли;
			КонецЕсли;
			Возврат;
		КонецЕсли;
	КонецЦикла;

&НаСервере
Процедура ДобавитьКнопкуПодсказкиКПолю(ИмяПоля)
	
	Поле = Элементы.Найти(ИмяПоля);
	Если Поле = Неопределено Тогда
		Возврат;
	КонецЕсли;
	Если Элементы.Найти("СправкаПоля" + ИмяПоля) <> Неопределено Тогда
		Возврат;
	КонецЕсли;
	
	ТекстПодсказкиПоля = "";
	Попытка
		ТекстПодсказкиПоля = СокрЛП(Строка(Поле.Подсказка));
	Исключение
		ТекстПодсказкиПоля = "";
	КонецПопытки;
	Если ПустаяСтрока(ТекстПодсказкиПоля) Тогда
		Возврат;
	КонецЕсли;
	
	Родитель = Поле.Родитель;
	Если Родитель = Неопределено Тогда
		Возврат;
	КонецЕсли;
	
	Если ТипЗнч(Родитель) = Тип("ТаблицаФормы") Тогда
		Возврат;
	КонецЕсли;
	
	Если ТипЗнч(Родитель) = Тип("ГруппаФормы")
		И Родитель.Вид = ВидГруппыФормы.КоманднаяПанель Тогда
		Возврат;
	КонецЕсли;
	
	ИмяГруппы = "ОберткаПодсказки" + ИмяПоля;
	Если Элементы.Найти(ИмяГруппы) <> Неопределено Тогда
		Возврат;
	КонецЕсли;
	
	Группа = Элементы.Добавить(ИмяГруппы, Тип("ГруппаФормы"), Родитель);
